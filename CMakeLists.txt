# =============================================================================
#  MiniCV 项目的编译说明书（CMakeLists.txt）
# -----------------------------------------------------------------------------
#  💡 一句话理解 CMake：
#  如果把写 C++ 代码比作「备菜」，那这个文件就是「菜谱」，告诉电脑怎么把你写的
#  一堆 .cpp 和 .h 文件打包成一个能直接双击/运行的程序（这里叫 minicv）。
#
#  🚀 平时在终端里怎么用？（复制粘贴就能跑）
#
#  【情况 A：平时运行 & 测运行速度（Release 模式，跑得快）】
#    1. 生成菜谱： cmake -B build-release -DCMAKE_BUILD_TYPE=Release
#    2. 开始编译： cmake --build build-release -j
#    3. 运行程序： ./build-release/minicv
#
#  【情况 B：程序崩溃了 / 查内存Bug（Debug 模式，带内存侦探，跑得慢）】
#    1. 生成菜谱： cmake -B build-asan -DCMAKE_BUILD_TYPE=Debug
#    2. 开始编译： cmake --build build-asan -j
#    3. 运行程序： ./build-asan/minicv
# =============================================================================


# -----------------------------------------------------------------------------
# 第 1 步：规定 CMake 软件的最低版本 & 给项目起个名字
# -----------------------------------------------------------------------------
# 要求电脑上的 CMake 版本至少是 3.16（必须在第一行）。
cmake_minimum_required(VERSION 3.16)

# 给项目起名叫 MiniCV。
# LANGUAGES CXX 的意思是：「这是一个纯 C++ 项目」（CXX 就是 C++ 的意思）。
project(MiniCV LANGUAGES CXX)


# -----------------------------------------------------------------------------
# 第 2 步：规定使用 C++17 标准
# -----------------------------------------------------------------------------
# 就像英语有四六级，C++ 也有 C++11、C++17、C++20 等版本。这里统一用 C++17。
set(CMAKE_CXX_STANDARD 17)           # 使用 C++17 标准
set(CMAKE_CXX_STANDARD_REQUIRED ON)  # 强制要求：如果编译器太老不支持 C++17，直接报错提醒我
set(CMAKE_CXX_EXTENSIONS OFF)        # 只用标准的 C++17，禁用编译器自己私加的「方言」语法


# -----------------------------------------------------------------------------
# 第 3 步：设置默认的打包模式（防呆设计）
# -----------------------------------------------------------------------------
# 编译一般分两种模式：
#   - Release（发布模式）：编译器会全力优化代码，跑得飞快，用来测性能。
#   - Debug（调试模式）  ：不优化，但会塞入很多检查工具，用来抓 Bug。
# 如果你在终端忘了写 -DCMAKE_BUILD_TYPE=...，下面这几行会自动帮你选 Release 模式。
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "编译模式: Debug 或 Release" FORCE)
endif()


# -----------------------------------------------------------------------------
# 第 4 步：去电脑里找 OpenMP（多线程加速工具）
# -----------------------------------------------------------------------------
# 我们的项目要用 OpenMP 来让 CPU 多核同时干活。
# REQUIRED 的意思是「必须找到」——如果电脑没装 OpenMP，现在就报错，别等编译到一半再崩。
find_package(OpenMP REQUIRED)


# -----------------------------------------------------------------------------
# 第 5 步：把 src/ 文件夹里所有的 .cpp 代码文件都找出来
# -----------------------------------------------------------------------------
# 自动去 src/ 文件夹（包括里面的子文件夹）搜集所有 .cpp 文件，存到 MINICV_SOURCES 这个列表里。
# 好处：以后你在 src/ 里新建了 .cpp 文件，不用每次都手动来改这个文件，CMake 会自动发现它。
file(GLOB_RECURSE MINICV_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")

# 如果 src/ 文件夹是空的，给一句看得懂的中文报错。
if(NOT MINICV_SOURCES)
  message(FATAL_ERROR "在 src/ 文件夹下没有找到任何 .cpp 文件！请先把源码放进 src/（比如 src/main.cpp）。")
endif()


# -----------------------------------------------------------------------------
# 第 6 步：把所有 .cpp 打包成最终的可执行程序 minicv
# -----------------------------------------------------------------------------
# 告诉 CMake：「请把上面找到的所有 .cpp 文件编译好，合成一个叫 minicv 的可执行程序」。
add_executable(minicv ${MINICV_SOURCES})


# -----------------------------------------------------------------------------
# 第 7 步：告诉编译器去哪里找 .h 头文件
# -----------------------------------------------------------------------------
# 当你在代码里写 #include "border/BorderPolicy.h" 时，编译器默认不知道去哪找。
# 这一行就是告诉它：「请去项目的 include/ 文件夹里找头文件」。
target_include_directories(minicv PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")


# -----------------------------------------------------------------------------
# 第 8 步：开启代码警告、性能优化、内存 Bug 检查
# -----------------------------------------------------------------------------
# 给 minicv 程序加上编译参数：
target_compile_options(minicv PRIVATE
  # 1) 【任何时候都开启】严格的代码体检（警告三件套）
  #    比如你定义了变量却没用、if 条件写错了，编译器都会发黄牌警告你。
  -Wall -Wextra -pedantic

  # 2) 【只在 Release 模式下开启】性能优化
  #    $<CONFIG:Release> 的意思是「只有当前是 Release 模式时，才加上后面的 -O2」。
  #    -O2 会让编译器自动优化你的代码，运行速度能快好几倍（注意是大写字母 O，不是数字 0）。
  "$<$<CONFIG:Release>:-O2>"

  # 3) 【只在 Debug 模式下开启】查内存 Bug 专用套装
  #    -g                        ：记住每一行代码的位置，报错时能精确告诉你死在第几行
  #    -O0                       ：关闭优化（字母 O + 数字 0），防止编译器把代码改得面目全非，方便调试
  #    -fsanitize=address        ：开启「内存侦探」(ASan)，数组越界、内存忘释放时立刻报警
  #    -fno-omit-frame-pointer   ：让报错时的函数调用记录更完整、更好读
  "$<$<CONFIG:Debug>:-g;-O0;-fsanitize=address;-fno-omit-frame-pointer>"
)

# 「内存侦探」(ASan) 不仅编译时要开，最后把文件拼成程序（链接）时也要加一次才能工作：
target_link_options(minicv PRIVATE "$<$<CONFIG:Debug>:-fsanitize=address>")


# -----------------------------------------------------------------------------
# 第 9 步：把 OpenMP 多线程库真正装进 minicv 程序里
# -----------------------------------------------------------------------------
# 第 4 步只是「找到了」OpenMP，这一步才是真正把它「组装」进我们的 minicv 程序里。
# 加上这一行后，你代码里的 #pragma omp parallel for 才能真正启用多线程。
target_link_libraries(minicv PRIVATE OpenMP::OpenMP_CXX)
